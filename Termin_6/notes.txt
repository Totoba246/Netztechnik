send:
    byte[] msgBuffer;
    send(FRAME_DELIMITER);
    for(int i = 0; i < msgBuffer.length; i++) {
        if(msgBuffer[i] == FRAME_DELIMITER || msgBuffer[i] == ESCAPE) {
            //Byte stuffing
            send(ESCAPE);
            send(msgBuffer[i]);
        }else {
            send(msgBuffer[i]);
        }
           
    }
    send(FRAME_DELIMITER);
